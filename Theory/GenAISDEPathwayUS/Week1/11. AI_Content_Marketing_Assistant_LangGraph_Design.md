# AI Content Marketing Assistant - Multi-Agent LangGraph Architecture

## Project Overview

**Project Name:** ContentAlchemy - Intelligent Multi-Format Content Creation System

**Description:** A sophisticated multi-agent content creation system that democratizes content creation by enabling intelligent content generation across multiple formats. The system solves content volume pressure, research bottlenecks, and resource constraints for small businesses and solo creators by providing research-first, SEO-optimized content generation.

**Key Objectives:**
- Enable multi-format content creation (blogs, LinkedIn posts, social content, images)
- SEO optimization with keyword research and metadata generation
- Research-first workflows with web integration
- Brand voice consistency across all formats
- Image generation with DALL-E integration
- Quality validation and improvement suggestions
- Error handling and API fallback mechanisms
- Multi-turn conversation support for iterative refinement

## Agent Specifications

### Agent 1: Query Handler Agent
**Purpose:** Routes user requests to appropriate specialized agents and manages conversation flow.

**Inputs:**
- User query/request (natural language)
- Conversation history
- Available content formats
- User preferences
- Content guidelines

**Outputs:**
- Routing decision
- Primary agent(s) to execute
- Secondary agents (if needed)
- Processing parameters
- Quality expectations

**Behavior:**
- Analyzes user intent
- Identifies content type requested
- Determines resource requirements
- Selects optimal agent combination
- Sets quality parameters
- Manages multi-turn conversations
- Maintains context across turns

**Route Matrix:**
```
Request Type → Primary Agent(s) → Parameters
─────────────────────────────────────────────
"Blog post" → Blog Writer + SEO Optimizer
"LinkedIn post" → Social Writer + Strategist
"Research brief" → Deep Research + Strategist
"Product copy" → Blog Writer + SEO
"Social campaign" → Deep Research + Social Writer + Strategist
"Multiple formats" → All agents (coordinated)
```

## State TypedDict Definition

```python
from typing import TypedDict, Optional, List, Dict, Any
from datetime import datetime

class ResearchData(TypedDict):
    """Research findings and data"""
    summary: str
    key_findings: List[str]
    statistics: List[Dict[str, Any]]
    sources: List[Dict[str, Any]]
    competitive_analysis: Dict[str, Any]
    trends: List[str]

class ContentPiece(TypedDict):
    """Individual content output"""
    format: str  # "blog", "linkedin", "twitter", "image", etc.
    content: str
    metadata: Dict[str, Any]
    quality_score: float
    created_at: datetime

class BrandVoiceGuidelines(TypedDict):
    """Brand consistency parameters"""
    tone: List[str]  # e.g., ["professional", "friendly", "authoritative"]
    values: List[str]
    color_palette: List[str]
    style_examples: List[str]
    taboo_topics: List[str]
    do_nots: List[str]

class ContentMarketingState(TypedDict):
    """Complete state for Content Marketing LangGraph"""
    # User Input & Context
    user_query: str
    conversation_history: List[Dict[str, str]]
    user_id: str
    session_id: str
    
    # User Profile
    brand_name: str
    brand_guidelines: Optional[BrandVoiceGuidelines]
    target_audience: str
    business_type: str
    
    # Request Details
    requested_formats: List[str]  # ["blog", "linkedin", "image", etc.]
    topic: str
    keywords: List[str]
    
    # Query Processing
    detected_intent: str
    selected_agents: List[str]
    routing_confidence: float
    
    # Research Phase
    research_data: Optional[ResearchData]
    research_sources: List[Dict[str, Any]]
    research_timestamp: Optional[datetime]
    
    # SEO Parameters
    primary_keyword: str
    secondary_keywords: List[str]
    target_word_count: int
    seo_target: str  # e.g., "rank for keyword", "improve CTR"
    
    # Generated Content
    content_pieces: Dict[str, ContentPiece]  # format -> piece mapping
    blog_content: Optional[str]
    blog_metadata: Optional[Dict[str, Any]]
    linkedin_post: Optional[str]
    twitter_post: Optional[str]
    generated_images: List[Dict[str, Any]]
    
    # Quality Assessment
    quality_scores: Dict[str, float]  # format -> score
    improvement_suggestions: Dict[str, List[str]]  # format -> suggestions
    overall_quality_score: float
    
    # Strategy & Planning
    content_strategy: Optional[Dict[str, Any]]
    series_recommendations: Optional[List[Dict[str, Any]]]
    distribution_plan: Optional[Dict[str, Any]]
    
    # Content Composition
    series_id: Optional[str]
    series_position: Optional[int]
    related_content: List[str]
    
    # Final Output
    final_response: str
    formatted_content: Dict[str, Any]
    delivery_format: str  # "json", "markdown", "html", etc.
    
    # Metadata
    execution_path: List[str]
    error_messages: List[str]
    fallback_triggered: bool
    total_tokens_used: int
    processing_time_seconds: float
```

def query_handler(state: ContentMarketingState) -> ContentMarketingState:
    """
    Analyzes user query and routes to appropriate agents
    """
    # Parse user intent
    intent_analysis = analyze_intent(state['user_query'])
    state['detected_intent'] = intent_analysis['intent']
    state['requested_formats'] = intent_analysis['formats']
    state['topic'] = intent_analysis['topic']
    state['keywords'] = intent_analysis['keywords']
    
    # Determine agents needed
    agents = determine_agents_needed(state['requested_formats'])
    state['selected_agents'] = agents
    state['routing_confidence'] = intent_analysis['confidence']
    
    state['execution_path'].append('query_handler')
    return state
```

**Transitions:**
- HIGH_CONFIDENCE → execute_selected_agents
- LOW_CONFIDENCE → clarification_request
- MULTI_AGENT → multi_agent_coordinator

def deep_research_agent(state: ContentMarketingState) -> ContentMarketingState:
    """
    Conducts comprehensive research on topic
    """
    if 'research' not in state['selected_agents']:
        state['execution_path'].append('deep_research_skipped')
        return state
    
    research_result = conduct_research(
        topic=state['topic'],
        keywords=state['keywords'],
        target_audience=state['target_audience'],
        num_sources=8
    )
    
    state['research_data'] = research_result['research']
    state['research_sources'] = research_result['sources']
    state['research_timestamp'] = datetime.now()
    state['execution_path'].append('deep_research_agent')
    
    return state
```

**Transitions:**
- SUCCESS → dependent_agents (blog_writer, strategist)
- NO_RESULTS → research_fallback
- API_ERROR → research_fallback

def seo_blog_writer_agent(state: ContentMarketingState) -> ContentMarketingState:
    """
    Generates SEO-optimized blog post
    """
    if 'blog' not in state['requested_formats']:
        state['execution_path'].append('blog_writer_skipped')
        return state
    
    # Perform keyword research if not done
    if not state['primary_keyword']:
        keyword_research = research_keywords(state['topic'])
        state['primary_keyword'] = keyword_research['primary']
        state['secondary_keywords'] = keyword_research['secondary']
    
    blog_result = generate_seo_blog(
        topic=state['topic'],
        primary_keyword=state['primary_keyword'],
        secondary_keywords=state['secondary_keywords'],
        research_data=state['research_data'],
        target_word_count=state['target_word_count'],
        brand_voice=state['brand_guidelines']
    )
    
    state['blog_content'] = blog_result['blog_post']
    state['blog_metadata'] = blog_result['metadata']
    state['content_pieces']['blog'] = ContentPiece(
        format='blog',
        content=blog_result['blog_post'],
        metadata=blog_result['metadata'],
        quality_score=blog_result['seo_score'],
        created_at=datetime.now()
    )
    
    state['execution_path'].append('seo_blog_writer_agent')
    return state

def image_generation_agent(state: ContentMarketingState) -> ContentMarketingState:
    """
    Generates custom images for content
    """
    if 'image' not in state['requested_formats']:
        state['execution_path'].append('image_generator_skipped')
        return state
    
    # Generate image prompts from topic and research
    image_concepts = generate_image_concepts(
        topic=state['topic'],
        research_data=state['research_data'],
        brand_guidelines=state['brand_guidelines'],
        num_variations=3
    )
    
    generated_images = []
    for concept in image_concepts:
        image_result = generate_image(
            prompt=concept['optimized_prompt'],
            style=concept['style'],
            aspect_ratio='16:9',  # Blog standard
            quality='high'
        )
        generated_images.append({
            'url': image_result['url'],

            'alt_text': image_result['alt_text'],

        format='image',

        metadata={'count': len(generated_images)},
        quality_score=0.9,
        created_at=datetime.now()
    )
    
    state['execution_path'].append('image_generation_agent')
    return state

#### Node 7: Multi-Agent Coordinator
```python
def multi_agent_coordinator(state: ContentMarketingState) -> ContentMarketingState:
    """
    Orchestrates parallel execution of multiple agents
    """
    # Execute research first (needed by others)
    if 'research' in state['selected_agents']:
        state = deep_research_agent(state)
    
    # Execute format agents in parallel (conceptually)
    agents_to_run = [
        a for a in state['selected_agents'] 
        if a in ['blog', 'linkedin', 'twitter', 'image', 'strategist']
    ]
    
    for agent_name in agents_to_run:
        if agent_name == 'blog':
            state = seo_blog_writer_agent(state)
        elif agent_name == 'linkedin':
            state = linkedin_post_writer_agent(state)
        elif agent_name == 'image':
            state = image_generation_agent(state)
        elif agent_name == 'strategist':
            state = content_strategist_agent(state)
    
    state['execution_path'].append('multi_agent_coordinator')
    return state
```

**Transitions:**
- SUCCESS → quality_validator

#### Node 8: Quality Validator
```python
def quality_validator(state: ContentMarketingState) -> ContentMarketingState:
    """
    Validates quality of all generated content
    """
    for format_type, piece in state['content_pieces'].items():
        validation = validate_content(
            content=piece['content'],
            format=format_type,
            brand_guidelines=state['brand_guidelines'],
            quality_threshold=0.75
        )
        
        state['quality_scores'][format_type] = validation['score']
        state['improvement_suggestions'][format_type] = validation['suggestions']
    
    # Calculate overall quality
    state['overall_quality_score'] = sum(
        state['quality_scores'].values()
    ) / len(state['quality_scores']) if state['quality_scores'] else 0
    
    state['execution_path'].append('quality_validator')
    return state
```

**Transitions:**
- HIGH_QUALITY (> 0.8) → format_response
- MEDIUM_QUALITY (0.6-0.8) → auto_improvement
- LOW_QUALITY (< 0.6) → content_regeneration

#### Node 9: Auto-Improvement
```python
def auto_improvement(state: ContentMarketingState) -> ContentMarketingState:
    """
    Automatically improves lower-quality content
    """
    for format_type, suggestions in state['improvement_suggestions'].items():
        if state['quality_scores'][format_type] < 0.8:
            improved_content = improve_content(
                content=state['content_pieces'][format_type]['content'],
                suggestions=suggestions,
                format_type=format_type
            )
            
            state['content_pieces'][format_type]['content'] = improved_content
    
    state['execution_path'].append('auto_improvement')
    return state
```

**Transitions:**
- SUCCESS → quality_validator (re-validate)
- MAX_ITERATIONS → format_response

def format_response(state: ContentMarketingState) -> ContentMarketingState:
    """
    Formats final response for delivery
    """
    formatted_output = {
        'content': {
            format_type: piece['content']
            for format_type, piece in state['content_pieces'].items()
        },
        'metadata': {
            format_type: piece['metadata']
            for format_type, piece in state['content_pieces'].items()
        },
        'quality_scores': state['quality_scores'],
        'quality_analysis': state['improvement_suggestions'],
        'strategy': state['content_strategy'],
        'series_recommendations': state['series_recommendations'],
        'distribution_plan': state['distribution_plan'],
        'sources': state['research_sources'] if state['research_data'] else []
    }
    
    state['final_response'] = json.dumps(formatted_output, indent=2)
    state['formatted_content'] = formatted_output
    
    state['execution_path'].append('format_response')
    return state
```

from langgraph.graph import StateGraph, END

def create_content_marketing_graph():
    graph = StateGraph(ContentMarketingState)
    
    # Add nodes
    graph.add_node("query_handler", query_handler)
    graph.add_node("deep_research", deep_research_agent)
    graph.add_node("blog_writer", seo_blog_writer_agent)
    graph.add_node("linkedin_writer", linkedin_post_writer_agent)
    graph.add_node("image_generator", image_generation_agent)
    graph.add_node("strategist", content_strategist_agent)
    graph.add_node("multi_agent", multi_agent_coordinator)
    graph.add_node("quality_validator", quality_validator)
    graph.add_node("auto_improve", auto_improvement)
    graph.add_node("format_response", format_response)
    
    # Entry point
    graph.set_entry_point("query_handler")
    
    # Query handler routing
    def route_query(state: ContentMarketingState) -> str:
        if state['routing_confidence'] < 0.6:
            return 'clarification'
        
        if len(state['selected_agents']) > 1:
            return 'multi_agent'
        else:
            # Route to specific agent
            if 'blog' in state['selected_agents']:
                return 'blog_writer'
            elif 'linkedin' in state['selected_agents']:
                return 'linkedin_writer'
            elif 'image' in state['selected_agents']:
                return 'image_generator'
            else:
                return 'query_handler'
    
    graph.add_conditional_edges(
        "query_handler",
        route_query,
        {
            "multi_agent": "multi_agent",
            "blog_writer": "blog_writer",
            "linkedin_writer": "linkedin_writer",
            "image_generator": "image_generator",
            "clarification": "query_handler"
        }
    )
    
    # Multi-agent coordinator
    graph.add_edge("multi_agent", "quality_validator")
    
    # Individual agents to quality validator
    graph.add_edge("blog_writer", "quality_validator")
    graph.add_edge("linkedin_writer", "quality_validator")
    graph.add_edge("image_generator", "quality_validator")
    
    # Quality routing
    def route_quality(state: ContentMarketingState) -> str:
        if state['overall_quality_score'] > 0.8:
            return 'format_response'
        else:
            return 'auto_improve'
    
    graph.add_conditional_edges(
        "quality_validator",
        route_quality,
        {
            "format_response": "format_response",
            "auto_improve": "auto_improve"
        }
    )
    
    # Auto-improvement feedback loop
    graph.add_edge("auto_improve", "quality_validator")
    
    # Format response to end
    graph.add_edge("format_response", END)
    
    return graph.compile()
```
