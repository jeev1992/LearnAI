# AI-Powered Email Assistant - Multi-Agent LangGraph Architecture

## Project Overview

**Project Name:** Intelligent Email Composition & Personalization System

**Description:** A productivity-focused multi-agent system that generates, personalizes, and validates email drafts in seconds. The system reduces email composition time from 15-20 minutes to under 2 minutes while maintaining context-aware writing, consistent tone, and personalization based on user profile and history.

**Key Objectives:**
- Rapid email draft generation (under 2 minutes)
- Multiple tone options (formal, casual, assertive, empathetic, etc.)
- Personalization using user profiles and communication history
- Grammar and tone validation
- Support for multiple email intents (outreach, follow-up, apology, information request)
- Real-time preview with live adjustments
- Fallback and error recovery with intelligent routing

```python
def input_parser(state: EmailAssistantState) -> EmailAssistantState:
    """
    Parses user input and extracts structured information
    - Extract intent, recipient, tone
    - Identify constraints
    - Validate completeness
    """
    parsed = extract_from_prompt(state['user_prompt'])
    state['parsed_intent'] = parsed['intent']
    state['recipient_info'] = parsed['recipient']
    state['selected_tone'] = parsed['tone']
    state['user_constraints'] = parsed['constraints']
    state['execution_path'].append('input_parser')
    return state
```

**Transitions:**
- INPUT_COMPLETE → intent_detection
- INPUT_INCOMPLETE → clarification_required

```python
def review_validator(state: EmailAssistantState) -> EmailAssistantState:
    """
    Validates grammar, tone, and coherence

    validation_result = validate_email(
        content=state['current_draft']['content'],
        subject=state['subject_line'],

        intent=state['detected_intent']

    state['grammar_score'] = validation_result['grammar_score']

    return state
```

## State Transitions Code

```python
from langgraph.graph import StateGraph, END
from langgraph.types import Send

def create_email_graph():
    graph = StateGraph(EmailAssistantState)
    
    # Add nodes
    graph.add_node("input_parser", input_parser)
    graph.add_node("intent_detection", intent_detection)
    graph.add_node("intent_clarification", intent_clarification)
    graph.add_node("tone_stylist", tone_stylist)
    graph.add_node("draft_writer", draft_writer)
    graph.add_node("personalization", personalization_agent)
    graph.add_node("review_validator", review_validator)
    graph.add_node("auto_improvement", auto_improvement)
    graph.add_node("error_correction", error_correction)
    graph.add_node("format_response", format_response)
    graph.add_node("memory_routing", memory_and_routing)
    
    # Entry point
    graph.set_entry_point("input_parser")
    
    # Input parser routing
    def route_input(state: EmailAssistantState) -> str:
        if state['parsed_intent'] and state['recipient_info']:
            return "intent_detection"
        else:
            return "intent_clarification"
    
    graph.add_conditional_edges(
        "input_parser",
        route_input,
        {
            "intent_detection": "intent_detection",
            "intent_clarification": "intent_clarification"
        }
    )
    
    # Intent detection routing
    def route_intent(state: EmailAssistantState) -> str:
        confidence = state['intent_confidence']
        if confidence > 0.8:
            return "tone_stylist"
        elif confidence < 0.5:
            return "intent_clarification"
        else:
            return "tone_stylist"  # Proceed with medium confidence
    
    graph.add_conditional_edges(
        "intent_detection",
        route_intent,
        {
            "tone_stylist": "tone_stylist",
            "intent_clarification": "intent_clarification"
        }
    )
    
    # Clarification re-routes
    graph.add_edge("intent_clarification", "intent_detection")
    
    # Main pipeline
    graph.add_edge("tone_stylist", "draft_writer")
    graph.add_edge("draft_writer", "personalization")
    graph.add_edge("personalization", "review_validator")
    
    # Quality routing
    def route_quality(state: EmailAssistantState) -> str:
        score = state['overall_quality_score']
        if score > 0.8:
            return "format_response"
        elif score > 0.6:
            return "auto_improvement"
        else:
            return "error_correction"
    
    graph.add_conditional_edges(
        "review_validator",
        route_quality,
        {
            "format_response": "format_response",
            "auto_improvement": "auto_improvement",
            "error_correction": "error_correction"
        }
    )
    
    # Improvement feedback loop
    graph.add_edge("auto_improvement", "review_validator")
    graph.add_edge("error_correction", "review_validator")
    
    # Final steps
    graph.add_edge("format_response", "memory_routing")
    graph.add_edge("memory_routing", END)
    
    return graph.compile()
```
